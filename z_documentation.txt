        NOTA: 
            Los middleware a partir de la versión 11 ya no se pueden estructurar de la siguiente manera
                public function __construct()
                {
                    $this->middleware('auth', ['except' => ['indice', 'autenticar']]);
                    $this->middleware('guest', ['only' => ['indice', 'autenticar']]);
                }
            Para ello se puede compensar con
                public function __construct()
                {
                    $this->middleware();
                }
                public static function middleware(): array
                {
                    return [
                        new Middleware(middleware: 'auth', except: ['indice', 'autenticar']),
                        new Middleware(middleware: 'guest', only: ['indice', 'autenticar']),
                    ];
                }
            O bien restringiendo las rutas
            Route::middleware(['auth', 'estudiante'])->group(function () {
                lista de rutas    
            });

Nota: Auth::User() user no hace referencia al modelo usuario, por lo tanto debe mantener su valor de clase "User" no cambiar

    public function getAuthPassword()

    ];
